# JniSharpGenerator
Jni wrappers generator for C# powered by [JavaNETInterface](https://github.com/ST0PL/JavaNETInterface).

## Usage
```
JniSharpGenerator <options>
Where possible options include:
  -a                               Generate field accessors
  -s                               Save classes to files
  -d <directory>                   Specify where to place generated source files
  --class-path <path>              Specify where to find user class files
  -classpath <path>                Specify where to find user class files
  -cp <path>                       Specify where to find user class files
```

### Class in java archive
```
JniSharpGenerator -a --class-path yourPackage.jar path.to.your.klass
```

```
JniSharpGenerator -a -classpath yourPackage.jar path.to.your.klass
```

```
JniSharpGenerator -a -cp yourPackage.jar path.to.your.klass
```
### Class in .class file
```
JniSharpGenerator -a path\to\klass.class
```

#### Prevent creation of package directory
```
JniSharpGenerator -a -s -d . path\to\klass.class
```

#### Sample.cs
```csharp
// <auto-generated />
using JavaNETInterface.Jni;
using JavaNETInterface.Wrappers;

namespace JniSharpGenerator.Sample;

public unsafe class Sample : JniObject
{
    public static JClass* Class { get; private set; }
    private static readonly Lock lockObject = new();
    private static bool initialized;

    private static JFieldID* value_FieldId;

    private static JMethodID* Sample_MethodId;
    private static JMethodID* main_MethodId;
    private static JMethodID* add_MethodId;
    private static JMethodID* nativeSharpMethod_MethodId;

    public Sample(JniEnv* env, JObject* obj) : base(env, obj)
        => Initialize(env);

    public static void Initialize(JniEnv* env)
    {
        lock(lockObject)
        {
            if(initialized)
                return;

            Class = env->FindClass("JniSharpGenerator/Sample/Sample");

            value_FieldId = env->GetStaticFieldID(Class, "value", "I");

            Sample_MethodId = env->GetMethodID(Class, "<init>", "()V");
            main_MethodId = env->GetStaticMethodID(Class, "main", "([Ljava/lang/String;)V");
            add_MethodId = env->GetStaticMethodID(Class, "add", "(II)I");
            nativeSharpMethod_MethodId = env->GetStaticMethodID(Class, "nativeSharpMethod", "()V");

            initialized = true;
        }
    }
    public static int GetValue(JniEnv* env)
    {
        var result = env->GetStaticIntField(Class, value_FieldId);
        return result;
    }

    public static void SetValue(JniEnv* env, int value)
    {
        env->SetStaticIntField(Class, value_FieldId, value);
    }

    /// <summary>
    /// To be documented
    /// </summary>
    public static Sample Sample_Constructor(JniEnv* env)
    {
        Initialize(env);
        var result = env->NewObject(Class, Sample_MethodId);
        return new(env, result);
    }
    /// <summary>
    /// To be documented
    /// </summary>
    /// <param name="arg0">java.lang.String[]</param>
    /// <returns>void</returns>
    public static void Main(JniEnv* env, JObjectArray* arg0)
    {
        JValue* args = stackalloc JValue[1];
        args[0].L = (JObject*)arg0;

        env->CallStaticVoidMethodA(Class, main_MethodId, args);
    }
    /// <summary>
    /// To be documented
    /// </summary>
    /// <returns>int</returns>
    public static int Add(JniEnv* env, int arg0, int arg1)
    {
        JValue* args = stackalloc JValue[2];
        args[0].I = arg0;
        args[1].I = arg1;

        var result = env->CallStaticIntMethodA(Class, add_MethodId, args);
        return result;
    }
    /// <summary>
    /// To be documented
    /// </summary>
    /// <returns>void</returns>
    public static void NativeSharpMethod(JniEnv* env)
    {
        env->CallStaticVoidMethod(Class, nativeSharpMethod_MethodId);
    }
}
```

#### JAgent.cs
```csharp
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using JavaNETInterface.Jni;
using JavaNETInterface.Jvmti;
using java.io;
using java.util;

namespace JniSharpGenerator.Sample;

public static unsafe class JAgent
{
    static JvmtiEventCallbacks callbacks = default;
    [UnmanagedCallersOnly(EntryPoint = "Agent_OnLoad")]
    public static int Agent_OnLoad(JavaVM* vm, byte* options, void* reserved)
        => Initialize(vm, options);

    [UnmanagedCallersOnly(EntryPoint = "Agent_OnAttach")]
    public static int Agent_OnAttach(JavaVM* vm, byte* options, void* reserved)
        => Initialize(vm, options);

    [UnmanagedCallersOnly(EntryPoint = "Agent_OnUnload")]
    public static void Agent_OnUnload(JavaVM* vm) { }

    private static int Initialize(JavaVM* vm, byte* options)
    {
        JvmtiEnv* jvmti = null;
        int res = vm->GetEnv((void**)&jvmti, JvmtiConstants.JVMTI_VERSION);
        if (res != JniConstants.JNI_OK || jvmti == null)
            return -1;

        callbacks.VMInit = &OnVMInit;

        fixed(JvmtiEventCallbacks* cptr = &callbacks)
            jvmti->SetEventCallbacks(cptr, sizeof(JvmtiEventCallbacks));

        jvmti->SetEventNotificationMode(JvmtiEventMode.ENABLE, JvmtiEvent.VM_INIT, null);

        return 0;
    }

    [UnmanagedCallersOnly(CallConvs = [typeof(CallConvCdecl)])]
    public static void OnVMInit(JvmtiEnv* jvmtiEnv, JniEnv* jniEnv, JObject* thread)
    {
        java.lang.System.Initialize(jniEnv);
        java.lang.System.GetOut(jniEnv).Println("[NativeAOT] JVM initialized.");
    }

    [UnmanagedCallersOnly(EntryPoint = "Java_JniSharpGenerator_Sample_Sample_nativeSharpMethod", CallConvs = [typeof(CallConvCdecl)])]
    public static void NativeSharpMethod(JniEnv* env, JClass* klass)
    {
        Sample.Initialize(env);

        PrintStream outStream = java.lang.System.GetOut(env);

        outStream.Print("[NativeAOT] Enter an integer: ");
        
        Scanner scanner = Scanner.Scanner2_Constructor(env, java.lang.System.GetIn(env));
        
        int param = scanner.NextInt();

        outStream.Println($"[NativeAOT] Result of {Sample.GetValue(env)} + {param} is {Sample.Add(env, Sample.GetValue(env), param)}");

        scanner.Close();
    }
}
```

#### Run.bat

The sample project includes a complete build and run pipeline. You can use the following script to compile the generator, generate wrappers, publish the C# NativeAOT agent, and run the Java application:
```
@echo off
:: 1. Build the generator tool
dotnet build ..\JniSharpGenerator -c Release

:: 2. Compile the Java source code
javac -d . Java\Sample.java

:: 3. Package the Java class into a JAR file
jar cvf Sample.jar JniSharpGenerator
rd /s /q JniSharpGenerator

:: 4. Generate C# JNI wrappers using the built generator
..\JniSharpGenerator\bin\x64\Release\JniSharpGenerator.exe -a -s -d . -cp Sample.jar JniSharpGenerator.Sample.Sample

:: 5. Publish the Sample project as a native binary (NativeAOT)
dotnet publish -c Release -r win-x64

:: 6. Run the Java application with the C# NativeAOT agent attached
java -agentpath:bin\x64\Release\JniSharpGenerator.Sample.dll -cp Sample.jar JniSharpGenerator.Sample.Sample

:: 7. Cleanup temporary JAR
del Sample.jar
```

# Installation & Setup
1. Make sure you have all [dependencies](https://github.com/ST0PL/JniSharpGenerator#dependencies) installed
2. Add `path\to\jdk-25\bin\` to your system **PATH** environment variable
3. Clone the repository with submodules:
   ```bash
   git clone --recurse-submodules https://github.com/ST0PL/JniSharpGenerator
   ```
4. Open `JniSharpGenerator.sln` in Visual Studio
5. Build the `JniSharpGenerator` project or run the `JniSharpGenerator.Sample` target
